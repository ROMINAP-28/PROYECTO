# ==============================================================================
# TRAVELINK - Programación de Tareas de Mantenimiento (Cron)
# ==============================================================================
# Instalar en el servidor Linux con: crontab mantenimiento/cron/crontab.txt
# Requiere variables de entorno en /etc/environment o cargadas en el script.
SHELL=/bin/bash
PATH=/usr/local/sbin:/usr/local/bin:/sbin:/bin:/usr/sbin:/usr/bin
MAILTO=""

# 1. Backup Diario de Base de Datos (Todos los días a las 02:00 AM)
0 2 * * * /bin/bash /opt/travelink/mantenimiento/scripts/backup.sh >> /var/log/travelink/backup-cron.log 2>&1

# 2. Prueba de Restauración Automatizada de Backup (Semanal: Domingos a las 02:30 AM)
30 2 * * 0 /bin/bash /opt/travelink/mantenimiento/scripts/restore-test.sh >> /var/log/travelink/restore-test-cron.log 2>&1

# 3. Purga y Limpieza de Logs Antiguos (Semanal: Domingos a las 03:00 AM)
0 3 * * 0 /bin/bash /opt/travelink/mantenimiento/scripts/limpiar-logs.sh /opt/travelink/logs 14 >> /var/log/travelink/limpieza-cron.log 2>&1

# 4. Healthcheck Continuo del Actuator (Cada 5 minutos)
*/5 * * * * /bin/bash /opt/travelink/mantenimiento/scripts/healthcheck.sh >> /var/log/travelink/healthcheck-cron.log 2>&1
